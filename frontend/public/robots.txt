# 무엇을 긁어 가도 되는지.
#
# 둘러보기를 계정 없이 열어 두면서 생긴 문제가 하나 있습니다. 글 하나에
# 딸린 동선 그림(/api/posts/*/map)은 우리가 구글에서 받아 오는 것이라
# 한 장이 곧 한 번의 호출입니다. 크롤러가 목록을 훑으며 그림까지 받아
# 가면 그것이 그대로 요금이 됩니다.
#
# 로그인 안 한 요청은 GoogleQuotaFilter 가 아이피마다 시간당으로 묶어
# 두므로 터지지는 않습니다. 다만 막을 수 있는 것을 굳이 쓰게 둘 이유는
# 없습니다.
#
# 사람이 여는 화면(/community)은 열어 둡니다. 남이 올린 길을 찾아
# 들어오는 것이 이 게시판이 있는 이유입니다.

User-agent: *

# 사람 몫이 아닌 것들
Disallow: /api/

# 계정이 있어야만 열리는 화면. 크롤러가 받아 가 봐야 빈 껍데기입니다.
Disallow: /trip/
Disallow: /travel/
Disallow: /money/
Disallow: /vote/
Disallow: /card/
Disallow: /admin/

# 초대 링크. 받은 사람만 열어야 하는 주소가 검색에 남으면 안 됩니다.
Disallow: /invite/

Allow: /


# 링크를 펼쳐 카드로 만들어 주는 것들.
#
# 위의 Disallow: /api/ 가 이들에게도 걸리면 og:image 로 적어 둔 동선 그림을
# 받지 못해, 카드가 글자만 남은 채로 뜹니다. 슬랙·트위터·페이스북은
# robots.txt 를 실제로 지키므로 여기서 그림 한 장만 따로 열어 줍니다.
#
# 위에서 걱정한 요금과는 다른 일입니다. 크롤러는 목록을 훑으며 전부 받아
# 가지만, 이들은 사람이 어딘가에 붙여 넣은 글 하나만 한 번 받아 갑니다.
#
# robots.txt 는 자기에게 맞는 묶음 하나만 읽고 나머지는 통째로 무시합니다.
# 그래서 위의 금지 목록을 여기에 다시 적습니다 — 빠뜨리면 그만큼 열립니다.
User-agent: Slackbot-LinkExpanding
User-agent: Slackbot
User-agent: Twitterbot
User-agent: facebookexternalhit
Allow: /api/posts/
Disallow: /api/
Disallow: /trip/
Disallow: /travel/
Disallow: /money/
Disallow: /vote/
Disallow: /card/
Disallow: /admin/
Disallow: /invite/
Allow: /
