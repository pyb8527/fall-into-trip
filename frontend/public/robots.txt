# 무엇을 긁어 가도 되는지.
#
# 둘러보기를 계정 없이 열어 두면서 생긴 문제가 하나 있습니다. 글 하나에
# 딸린 동선 그림(/api/posts/*/map)은 우리가 구글에서 받아 오는 것이라
# 한 장이 곧 한 번의 호출입니다. 크롤러가 목록을 훑으며 그림까지 받아
# 가면 그것이 그대로 요금이 됩니다.
#
# 로그인 안 한 요청은 GoogleQuotaFilter 가 아이피마다 시간당으로 묶어
# 두므로 터지지는 않습니다. 다만 막을 수 있는 것을 굳이 쓰게 둘 이유는
# 없습니다.
#
# 사람이 여는 화면(/community)은 열어 둡니다. 남이 올린 길을 찾아
# 들어오는 것이 이 게시판이 있는 이유입니다.

User-agent: *

# 사람 몫이 아닌 것들
Disallow: /api/

# 계정이 있어야만 열리는 화면. 크롤러가 받아 가 봐야 빈 껍데기입니다.
Disallow: /trip/
Disallow: /travel/
Disallow: /money/
Disallow: /vote/
Disallow: /card/
Disallow: /admin/

# 초대 링크. 받은 사람만 열어야 하는 주소가 검색에 남으면 안 됩니다.
Disallow: /invite/
